from datetime import timedelta
from django.utils import timezone

def VerifyOrderOTP(request):
    order_id = request.session.get('order_id')

    if not order_id:
        messages.error(request, "Session expired. Please order again.")
        return redirect('Customermain')

    order = get_object_or_404(buy, id=order_id)

    # OTP expiry check
    if timezone.now() > order.otp_created_at + timedelta(minutes=5):
        messages.error(request, "OTP expired.")
        order.delete()
        del request.session['order_id']
        return redirect('Customermain')

    if request.method == 'POST':
        entered_otp = request.POST.get('otp')

        if order.otp_attempts >= 3:
            messages.error(request, "Too many wrong attempts.")
            order.delete()
            del request.session['order_id']
            return redirect('Customermain')

        if entered_otp == order.otp:
            order.is_verified = True
            order.otp = None
            order.save()
            del request.session['order_id']
            messages.success(request, "Order confirmed!")
            return redirect('Customermain')
        else:
            order.otp_attempts += 1
            order.save()
            messages.error(request, "Invalid OTP")

    return render(request, 'verify_order_otp.html')
